cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2 CACHE STRING "Build only for the non-wireless Pico 2")
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to the official Raspberry Pi Pico SDK (tested: 2.2.0).")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(adxl1005_test C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(adxl1005_test main.c)
target_link_libraries(adxl1005_test pico_stdlib hardware_adc hardware_dma hardware_clocks)
target_compile_options(adxl1005_test PRIVATE -Wall -Wextra)
target_compile_definitions(adxl1005_test PRIVATE PICO_STDIO_DEFAULT_CRLF=0 PICO_STDIO_USB_STDOUT_TIMEOUT_US=250000)
pico_enable_stdio_usb(adxl1005_test 1)
pico_enable_stdio_uart(adxl1005_test 0)
pico_set_program_name(adxl1005_test "ADXL1005 tap recorder")
pico_set_program_version(adxl1005_test "1.0")
pico_add_extra_outputs(adxl1005_test)
